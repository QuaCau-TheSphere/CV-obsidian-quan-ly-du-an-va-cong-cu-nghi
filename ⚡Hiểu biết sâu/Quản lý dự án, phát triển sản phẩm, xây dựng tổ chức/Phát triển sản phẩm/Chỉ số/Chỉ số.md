Khái niệm:: [[Thước đo, đo lường, chỉ số, KPI]]
```dataview
LIST rows.file.link
FROM "⚡Hiểu biết sâu/Quản lý dự án, phát triển sản phẩm, xây dựng tổ chức/Phát triển sản phẩm/Chỉ số" 
GROUP BY split(file.folder, "/")[4]
WHERE file.name != this.file.name
```
