```dataview
LIST rows.file.link
FROM "⚡Hiểu biết sâu/Bản thể luận, nhận thức luận, phương pháp luận/Phương pháp luận/Định tính/Phỏng vấn"
GROUP BY split(file.folder, "/")[3]
WHERE file.name != this.file.name
```
